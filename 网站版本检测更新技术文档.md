## 技术架构概述

### 核心技术栈

- **Service Worker**: 用于版本检测、缓存控制和离线支持
- **版本号系统**: `V + YYYYMMDD.TotalCount` 格式
- **更新检测**: 多维度检测机制（Service Worker 状态、文件内容、版本号比较）
- **自动更新**: 检测到新版本后立即提示用户更新
- **缓存策略**: 网络优先 + 缓存备用，确保获取最新代码

### 架构特点

1. **多设备同步检测**: 所有设备都能检测到网站更新
2. **实时版本检测**: 页面加载时和定期检查更新
3. **强制更新机制**: 检测到新版本后强制用户更新
4. **缓存控制**: 通过版本号和缓存破坏参数确保获取最新代码
5. **更新日志**: 显示详细的更新内容，提升用户体验

## 版本号管理机制

### 版本号格式

```
V + YYYYMMDD + . + TotalCount
```

**示例**: `V251217.67`
- `V`: 版本标识符
- `251217`: 日期（2025年12月17日，年份取后2位）
- `67`: 总更新次数（累计，不按天重置）

### 版本号规则

1. **日期部分**: 使用当前日期，格式为 `YYMMDD`
2. **计数部分**: 累计总更新次数，从 `update-log.json` 统计
3. **唯一性**: 每次更新版本号必须递增
4. **兼容性**: 支持旧格式 `20251212.8`（自动转换）

### 版本号存储位置

需要在以下文件中同步更新版本号：

1. **`index.html`**: `const APP_VERSION = 'V251217.67';`
2. **`sw.js`**: `const VERSION = 'V251217.67';`
3. **`site.webmanifest`**: `"version": "V251217.67"`
4. **`update-log.json`**: 添加新版本条目

### 自动版本更新脚本

```javascript
// update-version.js
const fs = require('fs');
const path = require('path');

// 统计 update-log.json 中的总版本数
function countTotalVersions() {
  const logPath = path.join(__dirname, 'update-log.json');
  if (!fs.existsSync(logPath)) {
    return 0;
  }
  
  try {
    const logContent = fs.readFileSync(logPath, 'utf8');
    const versionPattern = /"(V\d{6}\.\d+|202\d{5}\.\d+)"/g;
    const matches = logContent.match(versionPattern);
    return matches ? matches.length : 0;
  } catch (err) {
    console.error('读取 update-log.json 失败:', err);
    return 0;
  }
}

// 获取新版本号
function getNewVersion() {
  const args = process.argv.slice(2);
  if (args.length > 0) {
    return args[0];
  }
  
  const now = new Date();
  const year = String(now.getFullYear()).slice(-2);
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  const dateStr = `${year}${month}${day}`;
  
  const totalVersions = countTotalVersions();
  const newVersionNumber = totalVersions + 1;
  
  return `V${dateStr}.${newVersionNumber}`;
}

const newVersion = getNewVersion();

// 更新 sw.js
const swPath = path.join(__dirname, 'sw.js');
let swContent = fs.readFileSync(swPath, 'utf8');
swContent = swContent.replace(
  /const VERSION = ['"](.*?)['"];/g,
  `const VERSION = '${newVersion}';`
);
fs.writeFileSync(swPath, swContent, 'utf8');

// 更新 index.html
const htmlPath = path.join(__dirname, 'index.html');
let htmlContent = fs.readFileSync(htmlPath, 'utf8');
htmlContent = htmlContent.replace(
  /const APP_VERSION = ['"](.*?)['"];/,
  `const APP_VERSION = '${newVersion}';`
);
const datePart = newVersion.replace(/^V/, '').split('.')[0];
htmlContent = htmlContent.replace(
  /href="\.\/css\/style\.css\?v=(\d+)"/,
  `href="./css/style.css?v=${datePart}"`
);
fs.writeFileSync(htmlPath, htmlContent, 'utf8');

// 更新 site.webmanifest
const manifestPath = path.join(__dirname, 'site.webmanifest');
let manifestContent = fs.readFileSync(manifestPath, 'utf8');
manifestContent = manifestContent.replace(
  /"version":\s*["'](.*?)["']/,
  `"version": "${newVersion}"`
);
const datePart = newVersion.replace(/^V/, '').split('.')[0];
manifestContent = manifestContent.replace(
  /"start_url":\s*["']\.\/index\.html\?v=(\d+)["']/,
  `"start_url": "./index.html?v=${datePart}"`
);
fs.writeFileSync(manifestPath, manifestContent, 'utf8');

console.log(`版本号更新完成！新版本: ${newVersion}`);
```

**使用方法**:
```bash
# 自动计算版本号
node update-version.js

# 手动指定版本号
node update-version.js V251217.68
```

## Service Worker 版本检测

### Service Worker 注册

```javascript
// index.html
const APP_VERSION = 'V251217.67';
window.APP_VERSION = APP_VERSION;

if ('serviceWorker' in navigator) {
  let registration = null;
  
  // 注册 Service Worker（添加版本号和时间戳查询参数）
  const swUrl = './sw.js?v=' + APP_VERSION + '&t=' + Date.now() + 
                '&nocache=' + Math.random() + '&force=' + Math.random();
  
  navigator.serviceWorker.register(swUrl, {
    scope: './',
    updateViaCache: 'none' // 禁用 Service Worker 脚本的 HTTP 缓存
  }).then((reg) => {
    registration = reg;
    console.log('Service Worker 注册成功，版本:', APP_VERSION);
    
    // 立即检查更新
    checkForUpdate(reg);
    
    // 监听 Service Worker 更新
    reg.addEventListener('updatefound', () => {
      const newWorker = reg.installing;
      if (!newWorker) return;
      
      newWorker.addEventListener('statechange', () => {
        if (newWorker.state === 'installed') {
          if (navigator.serviceWorker.controller) {
            // 有新版本已安装，提示更新
            showUpdateNotification();
          }
        }
      });
    });
    
    // 定期检查更新（每10秒）
    setInterval(() => {
      checkForUpdate(reg);
    }, 10000);
  });
}
```

### Service Worker 文件结构

```javascript
// sw.js
const VERSION = 'V251217.67';
const CACHE_NAME = `xhsnum-cache-v${VERSION}`;

// 需要缓存的资源列表
const CRITICAL_RESOURCES = [
  './',
  './index.html',
  './css/style.css',
  './js/app.js',
  './site.webmanifest',
  './update-log.json',
  './icon/icon-192.png',
  './icon/icon-512.png'
];

// 安装 Service Worker
self.addEventListener('install', (event) => {
  console.log('[SW] 安装中，版本:', VERSION);
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll(CRITICAL_RESOURCES).catch((err) => {
        console.warn('[SW] 部分资源缓存失败:', err);
      });
    })
  );
  // 立即激活新的 Service Worker
  self.skipWaiting();
});

// 激活 Service Worker
self.addEventListener('activate', (event) => {
  console.log('[SW] 激活中，版本:', VERSION);
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cacheName) => {
          // 删除旧版本的缓存
          if (cacheName !== CACHE_NAME) {
            console.log('[SW] 删除旧缓存:', cacheName);
            return caches.delete(cacheName);
          }
        })
      );
    })
  );
  // 立即控制所有客户端
  return self.clients.claim();
});
```

## 多设备更新检测机制

### 检测机制概述

使用多维度检测机制，确保所有设备都能检测到更新：

1. **Service Worker 状态检测**: 检查 `registration.waiting` 和 `registration.installing`
2. **文件内容检测**: 直接读取 `sw.js` 文件内容，解析版本号
3. **版本号比较**: 比较本地版本号和服务器版本号
4. **定期检查**: 每10秒自动检查一次更新

### 更新检测函数

```javascript
async function checkForUpdate(reg) {
  if (!reg) {
    console.warn('Service Worker 注册对象不存在');
    return;
  }
  
  try {
    // 强制更新检查（忽略缓存）
    await reg.update();
    console.log('更新检查完成');
    
    // 延迟检查，确保状态已更新
    await new Promise(resolve => setTimeout(resolve, 300));
    
    // 检查是否有等待中的新版本
    if (reg.waiting) {
      console.log('发现等待中的新版本');
      showUpdateNotification();
      return;
    }
    
    // 检查是否有正在安装的新版本
    if (reg.installing) {
      reg.installing.addEventListener('statechange', () => {
        if (reg.installing && reg.installing.state === 'installed' && 
            navigator.serviceWorker.controller) {
          showUpdateNotification();
        }
      });
    }
    
    // 检查当前 Service Worker 的版本（通过 fetch 获取 sw.js 内容）
    try {
      const cacheBuster = Date.now() + Math.random();
      const swResponse = await fetch('./sw.js?v=' + APP_VERSION + '&t=' + 
                                     cacheBuster + '&nocache=' + Math.random() + 
                                     '&force=' + Math.random(), {
        cache: 'no-store',
        headers: {
          'Cache-Control': 'no-cache, no-store, must-revalidate',
          'Pragma': 'no-cache'
        }
      });
      
      if (swResponse.ok) {
        const swText = await swResponse.text();
        const versionMatch = swText.match(/const VERSION = ['"]([^'"]+)['"]/) || 
                             swText.match(/VERSION\s*=\s*['"]([^'"]+)['"]/);
        
        if (versionMatch) {
          const serverVersion = versionMatch[1];
          console.log('版本比较:', {
            current: APP_VERSION,
            server: serverVersion,
            match: serverVersion === APP_VERSION
          });
          
          if (serverVersion !== APP_VERSION) {
            console.log('检测到版本不匹配，需要更新');
            showUpdateNotification();
          }
        }
      }
    } catch (err) {
      console.warn('检查 Service Worker 版本失败:', err);
    }
  } catch (err) {
    console.warn('更新检查失败:', err);
  }
}
```

### 检测触发时机

1. **页面加载时**: 立即检查更新
2. **定期检查**: 每10秒自动检查一次
3. **Service Worker 更新事件**: `updatefound` 事件触发时检查
4. **手动触发**: 用户可以通过按钮手动检查更新

## 立即更新机制

### 更新通知显示

async function showUpdateNotification() {
  // 检查是否已经有提示框在显示
  if (document.getElementById('updateOverlay')) {
    console.log('更新提示框已显示，跳过重复显示');
    return;
  }
  
  // 检查是否已经显示过通知
  const lastShownVersion = localStorage.getItem('update_notification_shown');
  if (lastShownVersion && !lastShownVersion.startsWith('V')) {
    // 兼容旧版本号格式
    localStorage.removeItem('update_notification_shown');
  }
  
  const currentLastShown = localStorage.getItem('update_notification_shown');
  if (currentLastShown === APP_VERSION && pendingUpdate === false) {
    console.log('当前版本已显示过更新提示，跳过');
    return;
  }
  
  // 标记有待处理的更新
  pendingUpdate = true;
  
  // 从更新日志文件读取更新内容
  let updateInfo = {
    title: '发现新版本',
    content: ['本次更新包含了一些改进和修复']
  };
  
  try {
    const response = await fetch('./update-log.json?v=' + APP_VERSION + 
                                 '&t=' + Date.now() + '&nocache=' + Math.random(), {

      headers: {
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Pragma': 'no-cache'
      }
    });
    if (response.ok) {
      const updateLog = await response.json();
      if (updateLog[APP_VERSION]) {
        updateInfo = updateLog[APP_VERSION];
      }
    }
  } catch (err) {
    console.warn('无法加载更新日志:', err);

        background: #1990FF;

### 更新执行流程

```javascript
async function handleUpdateConfirm(e) {
  e.preventDefault();
  e.stopPropagation();
  
  const btnUpdateConfirm = document.getElementById('btnUpdateConfirm');
  btnUpdateConfirm.disabled = true;
  btnUpdateConfirm.textContent = '更新中...';
  
  try {
    // 移除提示框
    const overlay = document.getElementById('updateOverlay');
    if (overlay && overlay.parentNode) {
      overlay.remove();
    }
    
    // 记录已显示通知
    localStorage.setItem('update_notification_shown', APP_VERSION);
    
    // 标记正在刷新
    refreshing = true;
    pendingUpdate = false;
    
    // 设置手动刷新标志
    sessionStorage.setItem('sw_manual_refresh', 'true');
    sessionStorage.setItem('sw_refreshing', 'true');
    sessionStorage.setItem('sw_refresh_time', Date.now().toString());
    
    // 通知 Service Worker 跳过等待
    if (registration && registration.waiting) {
      registration.waiting.postMessage({ type: 'SKIP_WAITING' });
    }
    
    // 清除所有缓存
    if (registration) {
      try {
        const cacheNames = await caches.keys();
        await Promise.all(cacheNames.map(name => caches.delete(name)));
        console.log('所有缓存已清除');
      } catch (err) {
        console.warn('清除缓存失败:', err);
      }
    }
    
    // 额外清除：强制清除所有可能的缓存
    try {
      if ('caches' in window) {
        const allCacheNames = await caches.keys();
        for (const cacheName of allCacheNames) {
          await caches.delete(cacheName);
        }
        console.log('额外缓存清除完成');
      }
    } catch (err) {
      console.warn('额外缓存清除失败:', err);
    }
    
    // 延迟一小段时间，确保标志已设置，然后刷新
    setTimeout(() => {
      window.location.reload();
    }, 50);
  } catch (err) {
    console.error('更新失败:', err);
    alert('更新失败，请手动刷新页面');
    refreshing = false;
    pendingUpdate = false;
  }
}
```

### 缓存破坏参数

### Service Worker 缓存策略

```javascript
// sw.js
self.addEventListener('fetch', (event) => {
  const { request } = event;
  const url = new URL(request.url);
  
  // 只处理同源请求
  if (url.origin !== location.origin) {
    return;
  }
  
  // 对于 HTML 和关键资源，使用网络优先策略
  if (request.method === 'GET' && (
    request.destination === 'document' ||
    request.url.includes('.css') ||
    request.url.includes('.js') ||
    request.url.includes('manifest')
  )) {
    // 特殊处理：app.js 强制从网络获取，不使用缓存
    if (request.url.includes('app.js')) {
      event.respondWith(
        fetch(request, {
          cache: 'no-store',
          headers: {
            'Cache-Control': 'no-cache, no-store, must-revalidate',
            'Pragma': 'no-cache'
          }
        }).then((response) => {
          // 如果网络请求成功，更新缓存
          if (response && response.status === 200) {
            const responseToCache = response.clone();
            caches.open(CACHE_NAME).then((cache) => {
              cache.put(request, responseToCache);
            });
          }
          return response;
        }).catch(() => {
          // 网络失败时，尝试从缓存获取
          return caches.match(request).then((cachedResponse) => {
            return cachedResponse || new Response('离线状态', { status: 503 });
          });
        })
      );
      return;
    }
    
    // 其他关键资源：网络优先
    event.respondWith(
      fetch(request)
        .then((response) => {
          // 如果网络请求成功，更新缓存
          if (response && response.status === 200) {
            const responseToCache = response.clone();
            caches.open(CACHE_NAME).then((cache) => {
              cache.put(request, responseToCache);
            });
          }
          return response;
        })
        .catch(() => {
          // 网络失败时，尝试从缓存获取
          return caches.match(request).then((cachedResponse) => {
            return cachedResponse || new Response('离线状态', { status: 503 });
          });
        })
    );
  } else {
    // 其他资源使用缓存优先策略
    event.respondWith(
      caches.match(request).then((cachedResponse) => {
        return cachedResponse || fetch(request).then((response) => {
          if (response && response.status === 200) {
            const responseToCache = response.clone();
            caches.open(CACHE_NAME).then((cache) => {
              cache.put(request, responseToCache);
            });
          }
          return response;
        });
      })
    );
  }
});
```

async function loadUpdateLog() {

    const response = await fetch('./update-log.json?v=' + APP_VERSION + 
                                 '&t=' + Date.now() + '&nocache=' + Math.random(), {
      cache: 'no-store',
      headers: {
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Pragma': 'no-cache'
      }
    });
    
    if (response.ok) {
      const updateLog = await response.json();
      if (updateLog[APP_VERSION]) {
        return updateLog[APP_VERSION];
      }
    }
  } catch (err) {
    console.warn('无法加载更新日志:', err);
  }
  
  return {
    title: '发现新版本',
    content: ['本次更新包含了一些改进和修复']
  };
}

### 1. 完整的更新检测流程

```javascript
// index.html
const APP_VERSION = 'V251217.67';
window.APP_VERSION = APP_VERSION;

let registration = null;
let refreshing = false;
let pendingUpdate = false;

if ('serviceWorker' in navigator) {
  // 注册 Service Worker
  const swUrl = './sw.js?v=' + APP_VERSION + '&t=' + Date.now() + 
                '&nocache=' + Math.random() + '&force=' + Math.random();
  
  navigator.serviceWorker.register(swUrl, {
    scope: './',
    updateViaCache: 'none'
  }).then((reg) => {
    registration = reg;
    
    // 立即检查更新
    checkForUpdate(reg);
    
    // 监听更新事件
    reg.addEventListener('updatefound', () => {
      const newWorker = reg.installing;
      if (!newWorker) return;
      
      newWorker.addEventListener('statechange', () => {
        if (newWorker.state === 'installed' && navigator.serviceWorker.controller) {
          showUpdateNotification();
        }
      });
    });
    
    // 定期检查更新
    setInterval(() => {
      checkForUpdate(reg);
    }, 10000);
  });
  
  // 监听 Service Worker 控制权变化
  navigator.serviceWorker.addEventListener('controllerchange', () => {
    if (refreshing) return;
    
    const isManualRefresh = sessionStorage.getItem('sw_manual_refresh') === 'true';
    if (isManualRefresh) {
      sessionStorage.removeItem('sw_manual_refresh');
      return;
    }
    
    window.location.reload();
  });
}
```

## 常见问题和解决方案

### 1. 多设备检测不到更新

**问题**: 某些设备检测不到新版本

**解决方案**:
- 使用多个缓存破坏参数（版本号、时间戳、随机数）
- 设置 `updateViaCache: 'none'` 禁用 Service Worker 缓存
- 使用 `cache: 'no-store'` 和 `Cache-Control` 头强制从网络获取
- 定期检查更新（每10秒）

```javascript
const swUrl = './sw.js?v=' + APP_VERSION + '&t=' + Date.now() + 
              '&nocache=' + Math.random() + '&force=' + Math.random();

navigator.serviceWorker.register(swUrl, {
  updateViaCache: 'none'
});
```

### 2. iOS Safari 更新检测问题

**问题**: iOS Safari 可能检测不到更新

**解决方案**:
- 使用多个检测机制（状态检测 + 文件内容检测）
- 增加检测延迟时间（300ms）
- 清除旧格式版本号记录

```javascript
// 延迟检查，确保状态已更新
await new Promise(resolve => setTimeout(resolve, 300));

// 兼容旧版本号格式
if (lastShownVersion && !lastShownVersion.startsWith('V')) {
  localStorage.removeItem('update_notification_shown');
}
```

### 3. Android 设备缓存问题

**问题**: Android 设备可能使用旧缓存

**解决方案**:
- 强制清除所有缓存
- 使用 `cache: 'no-store'` 和 `Cache-Control` 头
- 特殊处理 `app.js`，强制从网络获取

```javascript
// 特殊处理：app.js 强制从网络获取
if (request.url.includes('app.js')) {
  event.respondWith(
    fetch(request, {
      cache: 'no-store',
      headers: {
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Pragma': 'no-cache'
      }
    })
  );
}
```

### 4. 重复显示更新提示

**问题**: 更新提示重复显示

**解决方案**:
- 使用 `localStorage` 记录已显示的版本号
- 检查是否已有提示框在显示
- 使用 `pendingUpdate` 标志防止重复触发

```javascript
// 检查是否已经有提示框在显示
if (document.getElementById('updateOverlay')) {
  console.log('更新提示框已显示，跳过重复显示');
  return;
}

// 检查是否已经显示过通知
const lastShownVersion = localStorage.getItem('update_notification_shown');
if (lastShownVersion === APP_VERSION && pendingUpdate === false) {
  console.log('当前版本已显示过更新提示，跳过');
  return;
}
```

### 5. 刷新循环问题

**问题**: 更新后页面不断刷新

**解决方案**:
- 使用 `sessionStorage` 标记手动刷新
- 在 `controllerchange` 事件中检查刷新标志
- 设置刷新时间戳，防止无限循环

```javascript
// 设置手动刷新标志
sessionStorage.setItem('sw_manual_refresh', 'true');
sessionStorage.setItem('sw_refreshing', 'true');
sessionStorage.setItem('sw_refresh_time', Date.now().toString());

// 监听控制权变化
navigator.serviceWorker.addEventListener('controllerchange', () => {
  if (refreshing) return;
  
  const isManualRefresh = sessionStorage.getItem('sw_manual_refresh') === 'true';
  if (isManualRefresh) {
    sessionStorage.removeItem('sw_manual_refresh');
    return;
  }
  
  window.location.reload();
});
```
